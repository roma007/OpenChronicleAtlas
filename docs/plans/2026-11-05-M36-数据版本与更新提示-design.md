## 1. 数据版本生成

在 `scripts/build-search-index.ts` 中：

```ts
import { createHash } from 'node:crypto';

const fingerprint = createHash('sha256')
  .update(JSON.stringify(datasets.map((dataset) => ({
    manifest: dataset.manifest,
    routes: dataset.routes,
    events: dataset.events,
  }))))
  .digest('hex')
  .slice(0, 12);

const latestUpdatedAt = datasets
  .map((dataset) => dataset.manifest.updatedAt ?? '')
  .sort()
  .at(-1);

const dataVersion = `${latestUpdatedAt || new Date().toISOString().slice(0, 10)}-${fingerprint}`;
```

- 写入 `data/atlases/search-index.json`；
- 读取 `data/atlases/index.json`，写入 `dataVersion` 后回写；
- 只有 manifest/events/routes 变化才会改变哈希。

```ts
export function isDataVersionNewer(
  current: string | null | undefined,
  incoming: string | null | undefined,
): boolean {
  if (!incoming || !current) return false;
  return incoming !== current;
}
```
